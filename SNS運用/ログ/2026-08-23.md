```js
const ce = document.querySelector('div[data-testid^="tweetTextarea"] [contenteditable="true"]');
['pointerdown','mousedown','pointerup','mouseup','click'].forEach(t=>ce.dispatchEvent(new MouseEvent(t,{bubbles:true,cancelable:true,view:window})));
ce.focus();   // → これで確実にフォーカスできる。送信ボタンも同じ方式で押せる
```
